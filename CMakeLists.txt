cmake_minimum_required(VERSION 3.24)
project(Chuan VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(chuan_core src/core.cpp)
target_include_directories(chuan_core PUBLIC src)
target_compile_definitions(chuan_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(chuan_core PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(chuan_core PUBLIC ws2_32 iphlpapi bcrypt ole32 shell32)
add_executable(Chuan WIN32 src/app.cpp src/app.rc)
target_compile_options(Chuan PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/EHsc>)
target_link_libraries(Chuan PRIVATE chuan_core d2d1 dwrite comctl32 comdlg32 shlwapi dwmapi uxtheme)
target_link_options(Chuan PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_compile_options(core_tests PRIVATE /W4 /utf-8 /EHsc)
  target_link_libraries(core_tests PRIVATE chuan_core)
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(ui_tests tests/ui_tests.cpp src/app.rc)
  target_compile_options(ui_tests PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/EHsc>)
  target_link_libraries(ui_tests PRIVATE chuan_core d2d1 dwrite comctl32 comdlg32 shlwapi dwmapi uxtheme)
  target_link_options(ui_tests PRIVATE /MANIFEST:NO)
  add_test(NAME ui_idle_redraw COMMAND ui_tests "${CMAKE_CURRENT_BINARY_DIR}/ui-qa")
endif()
