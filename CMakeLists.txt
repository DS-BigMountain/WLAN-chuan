cmake_minimum_required(VERSION 3.24)
project(Chuan VERSION 2.2.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
function(chuan_medium_integrity target)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND icacls "$<TARGET_FILE:${target}>" /setintegritylevel M
    COMMENT "设置可执行文件的中完整性标签"
    VERBATIM)
endfunction()
add_library(chuan_core src/core.cpp src/third_party/qrcodegen/qrcodegen.cpp)
target_include_directories(chuan_core PUBLIC src)
target_compile_definitions(chuan_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000003)
target_compile_options(chuan_core PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(chuan_core PUBLIC ws2_32 iphlpapi bcrypt ole32 shell32 advapi32 psapi)
add_executable(Chuan WIN32 src/app.cpp src/app.rc)
set_property(SOURCE src/app.rc PROPERTY OBJECT_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/src/web/index.html;${CMAKE_CURRENT_SOURCE_DIR}/src/web/style.css;${CMAKE_CURRENT_SOURCE_DIR}/src/web/app.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/files.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/transfers.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/sha256.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/session.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/zip_store.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/zip_worker.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/approvals.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/hash_file.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/hash_worker.js;${CMAKE_CURRENT_SOURCE_DIR}/src/web/zip_stream_worker.js")
target_compile_options(Chuan PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/utf-8;/EHsc>)
target_link_libraries(Chuan PRIVATE chuan_core d2d1 dwrite comctl32 comdlg32 shlwapi dwmapi uxtheme)
target_link_options(Chuan PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
chuan_medium_integrity(Chuan)
